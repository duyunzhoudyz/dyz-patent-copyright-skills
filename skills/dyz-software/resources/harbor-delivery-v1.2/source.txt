[core.py:024] def validate_records(records, strict=False):

[core.py:026]     if not isinstance(records, list):

[core.py:028]     if len(records) > MAX_RECORDS:
[core.py:029]         raise InputError("too_many_records")
[core.py:030]     cleaned = []
[core.py:031]     seen_ids = set()
[core.py:032]     for record in records:
[core.py:033]         if not isinstance(record, dict):
[core.py:034]             raise InputError("record_must_be_object")
[core.py:035]         if not all(field in record for field in FIELDS):
[core.py:036]             raise InputError("missing_field")
[core.py:037]         if strict and set(record) != set(FIELDS):
[core.py:038]             raise InputError("unknown_field")
[core.py:039]         record_id, label, value = (record[field] for field in FIELDS)
[core.py:040]         if not isinstance(record_id, str) or not record_id.strip():

[core.py:042]         if not isinstance(label, str) or not label.strip():

[core.py:044]         try:
[core.py:045]             (record_id + label).encode("utf-8")
